USE ecommerce_analytics;

-- 1. Total Revenue
SELECT
    SUM(final_amount) AS total_revenue
FROM order_details;


-- 2. Total Orders
SELECT
    COUNT(DISTINCT order_id) AS total_orders
FROM orders;


-- 3. Category-wise Revenue
SELECT
    p.category,
    SUM(od.final_amount) AS total_revenue
FROM order_details od
JOIN products p
    ON od.product_id = p.product_id
GROUP BY p.category
ORDER BY total_revenue DESC;


-- 4. State-wise Revenue
SELECT
    c.state,
    SUM(od.final_amount) AS total_revenue
FROM order_details od
JOIN orders o
    ON od.order_id = o.order_id
JOIN customers c
    ON o.customer_id = c.customer_id
GROUP BY c.state
ORDER BY total_revenue DESC;


-- 5. Top 10 Products
SELECT
    p.product_name,
    SUM(od.final_amount) AS total_revenue
FROM order_details od
JOIN products p
    ON od.product_id = p.product_id
GROUP BY p.product_name
ORDER BY total_revenue DESC
LIMIT 10;


-- 6. Payment Mode Analysis
SELECT
    o.payment_mode,
    SUM(od.final_amount) AS total_revenue
FROM order_details od
JOIN orders o
    ON od.order_id = o.order_id
GROUP BY o.payment_mode
ORDER BY total_revenue DESC;


-- 7. Order Status Analysis
SELECT
    o.order_status,
    COUNT(*) AS total_orders
FROM orders o
GROUP BY o.order_status
ORDER BY total_orders DESC;


-- 8. Monthly Revenue
SELECT
    YEAR(o.order_date) AS year,
    MONTH(o.order_date) AS month,
    SUM(od.final_amount) AS monthly_revenue
FROM orders o
JOIN order_details od
    ON o.order_id = od.order_id
GROUP BY
    YEAR(o.order_date),
    MONTH(o.order_date)
ORDER BY
    year,
    month;


-- 9. Average Order Value
SELECT
    AVG(order_total) AS average_order_value
FROM (
    SELECT
        order_id,
        SUM(final_amount) AS order_total
    FROM order_details
    GROUP BY order_id
) AS order_summary;


-- 10. Customer-wise Revenue
SELECT
    c.customer_name,
    c.city,
    c.state,
    SUM(od.final_amount) AS total_revenue
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
JOIN order_details od
    ON o.order_id = od.order_id
GROUP BY
    c.customer_id,
    c.customer_name,
    c.city,
    c.state
ORDER BY total_revenue DESC;
